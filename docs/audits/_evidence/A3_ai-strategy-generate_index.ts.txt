   66: function stableStringify(value: unknown): string {
   67:   if (Array.isArray(value)) {
   68:     return `[${value.map((item) => stableStringify(item)).join(",")}]`;
   69:   }
   70:   if (value && typeof value === "object") {
   71:     const entries = Object.entries(value as Record<string, unknown>).sort(([a], [b]) => a.localeCompare(b));
   72:     const serialized = entries.map(([key, val]) => `${JSON.stringify(key)}:${stableStringify(val)}`).join(",");
   73:     return `{${serialized}}`;
   74:   }
   75:   return JSON.stringify(value);
   76: }

   78: async function sha256Hex(input: string) {

   80:   const digest = await crypto.subtle.digest("SHA-256", buffer);
   81:   return Array.from(new Uint8Array(digest))

   83:     .join("");
